import { useEffect, useState } from 'react'
import { customerBillingApi } from '../../api/api'
import { SidebarHeader } from './MetronicCatalog'
import { NavLink } from 'react-router-dom'
import '../../dashboard-palette.css'
import { useAuth } from '../../contexts/AuthContext'
import {
  LayoutDashboard, FileText, Zap, Bell,
  Building2, Network, User, Users, Upload, Globe, LogOut, Handshake, ShieldCheck, ReceiptText,
} from 'lucide-react'

export default function Sidebar({ open, mobile, hidden = false, panelRef, onClose, onNavigate }) {
  const { user, organization, logout } = useAuth()
  const platformAdmin = user?.platformRoles?.includes('C4G_ADMIN')
  const [billingAccess, setBillingAccess] = useState(null)
  useEffect(() => {
    setBillingAccess(null)
    if (!organization?.id || platformAdmin) return
    const controller = new AbortController()
    customerBillingApi(organization.id).access(controller.signal)
      .then(({data}) => { if (!controller.signal.aborted) setBillingAccess({organizationId:organization.id,userId:user?.id,allowed:data.canRead === true}) })
      .catch(() => {})
    return () => controller.abort()
  }, [organization?.id, user?.id, platformAdmin])
  const canReadBilling = billingAccess?.allowed && billingAccess.organizationId === organization?.id && billingAccess.userId === user?.id
  const groups = [
    ...(platformAdmin ? [{ label: 'Plataforma C4G', items: [
      { to: '/administracao/dashboard', icon: LayoutDashboard, label: 'Dashboard C4G' },
    ] }, { label: 'Oportunidades e catálogo', items: [
      { to: '/catalogo', icon: Globe, label: 'Catálogo C4G' },
      { to: '/administracao/oportunidades/nova', icon: FileText, label: 'Criar oportunidade' },
      { to: '/integracoes', icon: Globe, label: 'Fontes de oportunidades' },
      { to: '/administracao/importar', icon: Upload, label: 'Importar Portugal 2030' },
    ] }, { label: 'Administração e segurança', items: [
      { to: '/administracao/entidades', icon: Users, label: 'Entidades e utilizadores' },
      { to: '/administracao/faturacao', icon: ReceiptText, label: 'Faturação' },
      { to: '/administracao/logs', icon: FileText, label: 'Logs' },
      { to: '/conta/seguranca', icon: ShieldCheck, label: 'Segurança da conta' },
    ] }] : []),
    ...(organization ? [{ label: 'Trabalho diário', items: [
      { to: '/dashboard', icon: LayoutDashboard, label: 'Dashboard' },
      ...(!platformAdmin ? [{ to: '/catalogo', icon: Globe, label: 'Catálogo C4G' }] : []),
      { to: '/candidaturas', icon: FileText, label: 'Candidaturas' },
      { to: '/parceiros', icon: Handshake, label: 'Parceiros' },
      { to: '/matching', icon: Zap, label: 'Matching' },
      { to: '/notificacoes', icon: Bell, label: 'Alertas' },
    ] }, { label: 'Organização', items: [
      { to: '/perfil-organizacao', icon: Building2, label: 'Perfil da organização' },
      { to: '/perfil-organizacao/unidades', icon: Network, label: 'Departamentos / Unidades' },
      { to: '/perfil-organizacao/pessoas', icon: Users, label: 'Pessoas' },
      ...(canReadBilling ? [{ to: '/faturacao', icon: ReceiptText, label: 'Faturação' }] : []),
      { to: '/utilizadores', icon: ShieldCheck, label: 'Acessos' },
    ] }] : !platformAdmin ? [{ label: 'Trabalho diário', items: [
      { to: '/catalogo', icon: Globe, label: 'Catálogo C4G' },
    ] }] : []),
    ...(!platformAdmin || organization ? [{ label: 'A minha conta', items: [
      ...(organization ? [{ to: '/perfil', icon: User, label: 'Perfil pessoal' }] : []),
      ...(!platformAdmin ? [{ to: '/conta/seguranca', icon: ShieldCheck, label: 'Segurança da conta' }] : []),
    ] }] : []),
  ]

  return (
    <aside id="primary-navigation" ref={panelRef} onClickCapture={onNavigate} className={`sidebar${open ? ' sidebar-mobile-open' : ''}`}
      inert={(mobile && !open) || hidden ? '' : undefined} aria-hidden={(mobile && !open) || hidden ? true : undefined}
      role={mobile && open ? 'dialog' : undefined} aria-modal={mobile && open ? true : undefined}
      aria-label="Navegação principal">
      <SidebarHeader onClose={onClose} />
      <nav className="sidebar-nav">
        {groups.map(group => <section key={group.label} aria-label={group.label}>
          <div className="nav-divider"><span>{group.label}</span></div>
          {group.items.map(({ to, icon: Icon, label }) => (
            <NavLink key={to} to={to} end={to === '/perfil-organizacao'} className={({ isActive }) => 'nav-item' + (isActive ? ' active' : '')}>
              <Icon size={18} aria-hidden="true" />
              <span>{label}</span>
            </NavLink>
          ))}
        </section>)}
      </nav>

      <div className="sidebar-footer">
        <div className="sidebar-user">
          <div className="avatar">{user?.name?.[0]?.toUpperCase() || '?'}</div>
          <div className="sidebar-user-info">
            <span className="sidebar-user-name">{user?.name}</span>
            <span className="sidebar-user-role">{platformAdmin ? 'Administração C4G' : organization ? organization.profileLabel || 'Membro da organização' : 'Sem entidade selecionada'}</span>
          </div>
        </div>
        <button type="button" className="btn btn-logout logout-btn" onClick={logout}>
          <LogOut size={16} aria-hidden="true" />
          <span>Sair</span>
        </button>
      </div>
    </aside>
  )
}
